- [ ] **Step 1: Write the contrast helper** — `web/e2e/contrast.ts`

```ts
import type { Locator, Page } from "@playwright/test"

type RGBA = [number, number, number, number]

function parse(color: string): RGBA {
  const m = color.match(/rgba?\(([^)]+)\)/)
  if (!m) throw new Error(`unsupported colour ${color}`)
  const parts = m[1].split(/[\s,/]+/).filter(Boolean).map(Number)
  return [parts[0], parts[1], parts[2], parts[3] ?? 1]
}

function over(top: RGBA, bottom: RGBA): RGBA {
  const a = top[3]
  return [0, 1, 2].map((i) => top[i] * a + bottom[i] * (1 - a)).concat(1) as RGBA
}

function luminance([r, g, b]: RGBA): number {
  const lin = (c: number) => {
    const s = c / 255
    return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4
  }
  return 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b)
}

/** WCAG contrast ratio of colour a drawn over opaque colour b */
export function contrastRatio(a: string, b: string): number {
  const bottom = over(parse(b), [255, 255, 255, 1])
  const top = over(parse(a), bottom)
  const [l1, l2] = [luminance(top), luminance(bottom)].sort((x, y) => y - x)
  return (l1 + 0.05) / (l2 + 0.05)
}

/**
 * Focus the element with the keyboard, read its focus-ring colour (box-shadow ring colour,
 * or outline colour when there is no ring), and the nearest opaque background behind it.
 */
export async function focusRingContrast(page: Page, target: Locator): Promise<number> {
  await target.focus()
  const { ring, bg } = await target.evaluate((el) => {
    const style = getComputedStyle(el)
    const shadow = style.boxShadow
    const fromShadow = shadow && shadow !== "none" ? shadow.match(/rgba?\([^)]+\)/g) : null
    const ring = fromShadow?.[fromShadow.length - 1] ?? style.outlineColor
    let node: Element | null = el.parentElement
    let bg = "rgb(255, 255, 255)"
    while (node) {
      const c = getComputedStyle(node).backgroundColor
      if (c && !/rgba\(0, 0, 0, 0\)|transparent/.test(c)) {
        bg = c
        break
      }
      node = node.parentElement
    }
    return { ring, bg }
  })
  return contrastRatio(ring, bg)
}
```

Note:
- Tailwind v4 may emit `oklch()`/`color-mix()` in computed values. If `getComputedStyle` returns non-`rgb()` colours in Chromium, convert them in the page with a 1×1 canvas: set `ctx.fillStyle = color`, fill, then read `getImageData`. Add that path inside `evaluate` and keep the `rgb()` parser for the result.
- The ring may also be drawn as a `box-shadow` with several layers; the last colour layer is the ring.
- Verify the helper by checking its result against one hand-computed pair: `contrastRatio("rgb(0,0,0)", "rgb(255,255,255)")` is 21.
