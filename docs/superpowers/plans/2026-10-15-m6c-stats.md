# M6 round C: "สถิติของฉัน" stats tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the W6 "สถิติของฉัน" tab: four totals, a monthly table and a "ล้างสถิติ" reset, all computed locally from the `jobs` table.

**Architecture:** A pure Python function `core/stats.compute()` aggregates done jobs (bucketing months in UTC+07:00). A thin router `api/stats.py` serves `GET /api/stats` and `POST /api/stats/reset`; the reset writes only `statsResetAt` through a new `SettingsStore.reset_stats()`. The web adds a node-testable model `pages/settings/stats.ts`, a query module `lib/stats-api.ts` and a panel `pages/settings/stats-panel.tsx` mounted as a new settings tab.

**Tech Stack:** Python 3.13 + FastAPI + sqlite3 (`uv run pytest`); React 19 + TanStack Query + shadcn (base-ui) + sonner; Vitest; Playwright chromium.

**Spec:** `docs/superpowers/specs/2026-10-15-m6c-stats-design.md` (read it; §-numbers below point into it).

## Global Constraints

Copied from spec C1-C9. Every task holds all of them.

- C1 No network: nothing in `core/stats.py`, `api/stats.py` or the stats panel makes an outbound request; the panel calls only `/api/stats` and `/api/stats/reset`.
- C2 Privacy: no clip title, channel, path (`root`/`rel`), token or cookie data in the API response, logs or the report. The response holds numbers, kind keys, month keys and one timestamp only.
- C3 Read-only except `statsResetAt`: `POST /stats/reset` never deletes or edits jobs, videos, history, the queue or output files.
- C4 `PATCH /settings` still rejects `statsResetAt` (`E_BAD_REQUEST`); export and import still skip it; only `SettingsStore.reset_stats()` writes it.
- C5 `byMonth` buckets by the month of `finished_at` in fixed UTC+07:00, regardless of the machine's time zone.
- C6 A broken row never fails the endpoint (spec §1.4).
- C7 Thai UI copy only from spec §7 (repeated verbatim in Task 3). Buttons start with a verb, no "กรุณา", no "!". 640 px window; light and dark.
- C8 `video_id` is an opaque string; no YouTube id regex or YouTube wording in stats.
- C9 No `shell=True`; only `yttoolkit.platform` checks the OS; `yt_dlp` only in `core/extractor.py`; LF endings; every commit ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`; never commit `web/dist` (restore with `git checkout -- web/dist && git clean -fdq web/dist`); chromium e2e never creates real jobs and mocks every `/api/stats*` call; never use port 8787 for a real server; Playwright with `--workers=1`.

```python
"""สถิติของฉัน (M6 รอบ C spec §1, BRD FR-STAT-1..4, PRD :625)

คำนวณจากตาราง jobs ในเครื่องทุกครั้งที่เรียก ไม่มี network ไม่แตะไฟล์ (C1)
ผลมีแค่ตัวเลข ชนิด เดือน และเวลาเริ่มนับ ไม่มีชื่อคลิปหรือ path (C2) · แถวเสียข้ามเงียบ ๆ (C6)
"""

from __future__ import annotations

import json
from datetime import UTC, datetime, timedelta, timezone
from typing import TYPE_CHECKING, Any

if TYPE_CHECKING:
    from .db import Database

# เวลาไทยคงที่ (ไม่มี DST) ใช้จัดเดือน ไม่ขึ้นกับเขตเวลาของเครื่อง (C5, AC-STAT-01)
THAI = timezone(timedelta(hours=7))

# ชนิดผลลัพธ์ (core/jobs.OutputFile.kind) → 4 กลุ่มของ BRD (Pete Q1 = A) · ชนิดอื่นไม่นับ
KIND_GROUP = {
    "video": "video",
    "audio": "audio",
    "subtitle": "subtitle",
    "note": "subtitle",
    "image": "image",
    "archive": "image",
    "folder": "image",  # โฟลเดอร์ภาพคม = 1 ไฟล์ ขนาดทั้งโฟลเดอร์
}
GROUP_ORDER = ("video", "audio", "subtitle", "image")
MONTHS = 12


def _parse(value: object) -> datetime | None:
    if not isinstance(value, str) or not value:
        return None
    try:
        parsed = datetime.fromisoformat(value)
    except ValueError:
        return None
    return parsed if parsed.tzinfo else parsed.replace(tzinfo=UTC)


def _outputs(raw: object) -> list[tuple[str, int]]:
    """(กลุ่ม, ไบต์) ของแต่ละผลลัพธ์ที่นับได้ · JSON เสีย/ไม่ใช่ list = ไม่มีไฟล์"""
    if not isinstance(raw, str):
        return []
    try:
        items = json.loads(raw)
    except ValueError:
        return []
    if not isinstance(items, list):
        return []
    counted: list[tuple[str, int]] = []
    for item in items:
        if not isinstance(item, dict):
            continue
        group = KIND_GROUP.get(item.get("kind"))  # type: ignore[arg-type]
        if group is None:
            continue
        size = item.get("sizeBytes")
        ok = isinstance(size, int) and not isinstance(size, bool) and size >= 0
        counted.append((group, size if ok else 0))  # type: ignore[arg-type]
    return counted


def _month_key(moment: datetime) -> str:
    local = moment.astimezone(THAI)
    return f"{local.year:04d}-{local.month:02d}"


def _window(now: datetime) -> list[str]:
    local = now.astimezone(THAI)
    year, month = local.year, local.month
    keys: list[str] = []
    for _ in range(MONTHS):
        keys.append(f"{year:04d}-{month:02d}")
        year, month = (year, month - 1) if month > 1 else (year - 1, 12)
    return keys


def compute(db: Database, since: object, now: datetime) -> dict[str, Any]:
    start = _parse(since)
    rows = db.execute("SELECT video_id, finished_at, outputs_json FROM jobs WHERE status = 'done'")
    clips: set[str] = set()
    files = 0
    total = 0
    kinds: dict[str, list[int]] = {}
    months: dict[str, dict[str, Any]] = {}
    window = set(_window(now))
    for row in rows:
        finished = _parse(row["finished_at"])
        if finished is None or (start is not None and finished < start):
            continue
        video = str(row["video_id"])
        outputs = _outputs(row["outputs_json"])
        clips.add(video)
        files += len(outputs)
        total += sum(size for _, size in outputs)
        for group, size in outputs:
            entry = kinds.setdefault(group, [0, 0])
            entry[0] += 1
            entry[1] += size
        key = _month_key(finished)
        if key in window:
            month = months.setdefault(key, {"clips": set(), "files": 0, "bytes": 0})
            month["clips"].add(video)
            month["files"] += len(outputs)
            month["bytes"] += sum(size for _, size in outputs)
    by_kind = sorted(
        ({"kind": k, "files": v[0], "bytes": v[1]} for k, v in kinds.items()),
        key=lambda item: (-item["files"], -item["bytes"], GROUP_ORDER.index(item["kind"])),
    )
    by_month = [
        {"month": key, "clips": len(m["clips"]), "files": m["files"], "bytes": m["bytes"]}
        for key, m in sorted(months.items(), reverse=True)
    ]
    return {
        "since": since if start is not None else None,
        "clips": len(clips),
        "files": files,
        "totalBytes": total,
        "byKind": by_kind,
        "byMonth": by_month,
    }
```

```python
"""API สถิติของฉัน (M6 รอบ C spec §2.2, PRD :625-626): GET /stats · POST /stats/reset

คำนวณในเครื่องจากตาราง jobs ไม่มีคำขอออกนอกเครื่อง (FR-STAT-3) · POST ผ่าน token + Origin ของ LocalOnlyMiddleware
route เป็น `def` ธรรมดา (อ่าน SQLite) FastAPI จึงส่งไปรันใน threadpool · ไม่อ่าน body
"""

from __future__ import annotations

from datetime import UTC, datetime
from typing import TYPE_CHECKING

from fastapi import APIRouter
from fastapi.responses import JSONResponse

from .. import runstate
from ..core.stats import compute

if TYPE_CHECKING:
    from ..app import AppContext

NO_STORE = {"Cache-Control": "no-store"}


def build_stats_router(ctx: AppContext) -> APIRouter:
    router = APIRouter()
    db, settings = ctx.db, ctx.settings
    assert db is not None and settings is not None  # create_app เรียกต่อเมื่อมีทั้งสองอย่าง

    @router.get("/stats")
    def stats() -> JSONResponse:
        since = settings.get().get("statsResetAt")
        return JSONResponse(compute(db, since, datetime.now(UTC)), headers=NO_STORE)

    @router.post("/stats/reset")
    def stats_reset() -> JSONResponse:
        return JSONResponse({"since": settings.reset_stats(runstate.now_iso())}, headers=NO_STORE)

    return router
```

```tsx
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query"
import { Eraser } from "lucide-react"
import { useState } from "react"
import { toast } from "sonner"

import { ErrorDetails } from "@/components/error-report/error-details"
import { showErrorToast } from "@/components/error-report/error-toast"
import { PLAIN_TOAST } from "@/components/error-report/model"
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
} from "@/components/ui/alert-dialog"
import { Button } from "@/components/ui/button"
import { Empty, EmptyHeader, EmptyTitle } from "@/components/ui/empty"
import { Skeleton } from "@/components/ui/skeleton"
import { Spinner } from "@/components/ui/spinner"
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table"
import { fill, th } from "@/i18n/th"
import { formatBytes } from "@/lib/format"
import { reportRequest } from "@/lib/report"
import { resetStats, STATS_KEY, statsQuery } from "@/lib/stats-api"
import type { Stats } from "@/lib/types"

import { serverMessage } from "./general-form"
import { count, isEmpty, kindLabel, monthLabel, RESET_TOAST_ID, sinceLabel, topKind } from "./stats"

const t = th.stats

/** แท็บ "สถิติของฉัน" (M6 รอบ C spec §3.4, PRD :309 :326) · คำนวณในเครื่อง โหลดใหม่ทุกครั้งที่เปิดแท็บ */
export function StatsPanel() {
  const queryClient = useQueryClient()
  const stats = useQuery(statsQuery())
  const [confirmOpen, setConfirmOpen] = useState(false)

  const reset = useMutation({
    mutationFn: () => resetStats(),
    onSuccess: () => {
      setConfirmOpen(false)
      toast.success(t.resetDone, { ...PLAIN_TOAST, id: RESET_TOAST_ID })
      void queryClient.invalidateQueries({ queryKey: STATS_KEY })
    },
    onError: (error) => {
      setConfirmOpen(false)
      showErrorToast(serverMessage(error), { id: RESET_TOAST_ID, request: reportRequest(error, { context: "stats-reset" }) })
    },
  })

  if (stats.isPending) {
    return (
      <div className="grid grid-cols-2 gap-3 md:grid-cols-4" aria-busy="true">
        {[0, 1, 2, 3].map((i) => (
          <Skeleton key={i} className="h-20 w-full" />
        ))}
      </div>
    )
  }
  if (stats.isError) return <ErrorDetails error={stats.error} context="stats" />

  const data = stats.data
  return (
    <div className="flex flex-col gap-6">
      {isEmpty(data) ? (
        <Empty className="border py-12">
          <EmptyHeader>
            <EmptyTitle>{t.empty}</EmptyTitle>
          </EmptyHeader>
        </Empty>
      ) : (
        <>
          <Tiles stats={data} />
          <Months stats={data} />
        </>
      )}
      <div className="flex flex-col gap-1 text-sm text-muted-foreground">
        <p>{t.privacy}</p>
        {data.since && <p>{fill(t.since, { date: sinceLabel(data.since) })}</p>}
      </div>
      {!isEmpty(data) && (
        <div>
          <Button type="button" variant="ghost" disabled={reset.isPending} onClick={() => setConfirmOpen(true)}>
            <Eraser data-icon="inline-start" aria-hidden />
            {t.reset}
          </Button>
        </div>
      )}
      <AlertDialog
        open={confirmOpen}
        onOpenChange={(next) => {
          if (!next && !reset.isPending) setConfirmOpen(false)
        }}
      >
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>{t.resetTitle}</AlertDialogTitle>
            <AlertDialogDescription>{t.resetBody}</AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel disabled={reset.isPending}>{t.resetCancel}</AlertDialogCancel>
            <AlertDialogAction disabled={reset.isPending} onClick={() => reset.mutate()}>
              {reset.isPending && <Spinner />}
              {t.resetConfirm}
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  )
}

function Tiles({ stats }: { stats: Stats }) {
  const top = topKind(stats)
  const tiles: { label: string; value: string; sub?: string }[] = [
    { label: t.clips, value: count(stats.clips) },
    { label: t.files, value: count(stats.files) },
    { label: t.totalSize, value: formatBytes(stats.totalBytes) },
    { label: t.topKind, value: top ? kindLabel(top.kind) : "-", sub: top ? fill(t.topKindFiles, { n: count(top.files) }) : undefined },
  ]
  return (
    <dl className="grid grid-cols-2 gap-3 md:grid-cols-4" data-testid="stats-tiles">
      {tiles.map((tile) => (
        <div key={tile.label} className="flex min-w-0 flex-col gap-1 rounded-lg border p-4">
          <dt className="text-sm text-muted-foreground">{tile.label}</dt>
          <dd className="text-2xl font-semibold tabular-nums">{tile.value}</dd>
          {tile.sub && <dd className="text-sm text-muted-foreground">{tile.sub}</dd>}
        </div>
      ))}
    </dl>
  )
}

function Months({ stats }: { stats: Stats }) {
  return (
    <Table>
      <TableHeader>
        <TableRow>
          <TableHead scope="col">{t.month}</TableHead>
          <TableHead scope="col" className="text-right">{t.clips}</TableHead>
          <TableHead scope="col" className="text-right">{t.files}</TableHead>
          <TableHead scope="col" className="text-right">{t.size}</TableHead>
        </TableRow>
      </TableHeader>
      <TableBody>
        {stats.byMonth.map((row) => (
          <TableRow key={row.month}>
            <TableCell>{monthLabel(row.month)}</TableCell>
            <TableCell className="text-right tabular-nums">{count(row.clips)}</TableCell>
            <TableCell className="text-right tabular-nums">{count(row.files)}</TableCell>
            <TableCell className="text-right tabular-nums">{formatBytes(row.bytes)}</TableCell>
          </TableRow>
        ))}
      </TableBody>
    </Table>
  )
}
```
